qt_add_library(${PROJECT_NAME} STATIC)
qt_add_qml_module(${PROJECT_NAME}
    URI MTGScanner
    VERSION 1.0
    QML_FILES
        Main.qml
    SOURCES
        src/channel.hpp
        src/channeloptions.hpp
        src/channelmetrics.h
        src/channelmetrics.cpp
        src/framespersecond.hpp
)

qt_add_resources(${PROJECT_NAME} "app_icons"
    PREFIX "/${PROJECT_NAME}"
    FILES
        icons/app_titled.svg
        icons/app-window.svg
        icons/app.svg
        icons/check.svg
        icons/funnel-plus.svg
        icons/menu.svg
        icons/open-link.svg
        icons/open-link-60.svg
        icons/pause.svg
        icons/play.svg
        icons/plus.svg
        icons/square.svg
        icons/trash-2.svg
)

if (WIN32)
    set(WIN_RESOURCE_FILE "${CMAKE_CURRENT_SOURCE_DIR}/app_icon.rc")
endif()

target_sources(${PROJECT_NAME} PUBLIC ${WIN_RESOURCE_FILE})

target_link_libraries(${PROJECT_NAME} PUBLIC Qt6::QuickControls2 Qt6::Multimedia)
target_include_directories(${PROJECT_NAME} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)

add_subdirectory(Pages)
add_subdirectory(Components)

include(GenerateExportHeader)
target_include_directories(${PROJECT_NAME} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}) # find autogenerated header
generate_export_header(${PROJECT_NAME})
